INF = math.inf

def down(x: float, ulps: int = 1) -> float:
    for _ in range(ulps):
        x = math.nextafter(x, -INF)
    return x

def up(x: float, ulps: int = 1) -> float:
    for _ in range(ulps):
        x = math.nextafter(x, INF)
    return x

class IV:
    """Rigorous IEEE-754 outward-rounded interval [lo, hi].
    Every elementary arithmetic operation (+, -, *, /, sqrt) in IEEE-754 binary64 has
    rounding error <= 0.5 ULP, so stepping outward by 1 ULP via math.nextafter(..., +-inf)
    guarantees a strict outer enclosure of both the real-valued and IEEE-754 results.
    Transcendental sin/cos step outward by 2 ULPs to bound libm accuracy across platforms.
    """
    __slots__ = ("lo", "hi")

    def __init__(self, lo: float, hi: float = None):
        if hi is None:
            hi = lo
        assert lo <= hi, f"Invalid interval [{lo}, {hi}]"
        self.lo = float(lo)
        self.hi = float(hi)

    @classmethod
    def exact(cls, val: float):
        return cls(down(val, 1), up(val, 1))

    def __add__(self, other):
        if not isinstance(other, IV):
            other = IV(other, other)
        return IV(down(self.lo + other.lo), up(self.hi + other.hi))

    def __radd__(self, other):
        return self.__add__(other)

    def __sub__(self, other):
        if not isinstance(other, IV):
            other = IV(other, other)
        return IV(down(self.lo - other.hi), up(self.hi - other.lo))

    def __rsub__(self, other):
        if not isinstance(other, IV):
            other = IV(other, other)
        return other.__sub__(self)

    def __neg__(self):
        return IV(-self.hi, -self.lo)

    def __mul__(self, other):
        if not isinstance(other, IV):
            other = IV(other, other)
        p1 = self.lo * other.lo
        p2 = self.lo * other.hi
        p3 = self.hi * other.lo
        p4 = self.hi * other.hi
        return IV(down(min(p1, p2, p3, p4)), up(max(p1, p2, p3, p4)))

    def __rmul__(self, other):
        return self.__mul__(other)

    def __truediv__(self, other):
        if not isinstance(other, IV):
            other = IV(other, other)
        assert other.lo > 0.0 or other.hi < 0.0, f"Division by interval containing 0: [{other.lo}, {other.hi}]"
        q1 = self.lo / other.lo
        q2 = self.lo / other.hi
        q3 = self.hi / other.lo
        q4 = self.hi / other.hi
        return IV(down(min(q1, q2, q3, q4)), up(max(q1, q2, q3, q4)))

    @staticmethod
    def cos(iv_rad):
        assert iv_rad.lo >= 0.0 and iv_rad.hi <= math.pi
        return IV(down(math.cos(iv_rad.hi), 2), up(math.cos(iv_rad.lo), 2))

    @staticmethod
    def sin(iv_rad):
        assert iv_rad.lo >= 0.0 and iv_rad.hi <= math.pi
        s_lo = math.sin(iv_rad.lo)
        s_hi = math.sin(iv_rad.hi)
        lo_val = down(min(s_lo, s_hi), 2)
        if iv_rad.lo <= math.pi / 2.0 <= iv_rad.hi:
            hi_val = 1.0
        else:
            hi_val = up(max(s_lo, s_hi), 2)
        return IV(lo_val, hi_val)

    @staticmethod
    def min_list(ivs):
        return IV(min(x.lo for x in ivs), min(x.hi for x in ivs))

    @staticmethod
    def max_list(ivs):
        return IV(max(x.lo for x in ivs), max(x.hi for x in ivs))

    def __repr__(self):
        return f"[{self.lo:+.10f}, {self.hi:+.10f}]"

PI_IV = IV(down(math.pi), up(math.pi))
DEG2RAD = PI_IV / IV(180.0)
DT_IV = IV(1.3) / IV(60.0)
G_IV = IV(10.0)
F_IV = IV(550.0)

V0_DT_IV = V0_IV * DT_IV
G_DT2_IV = G_IV * DT_IV * DT_IV

def iv_eval_axis(uid: int, k: int, al_iv: IV, th_deg_iv: IV, kind: str, idx: int) -> IV:
    b = next(x for x in BLOCKS_SEED3000 if x["uid"] == uid)
    bx_iv, by_iv = IV.exact(b["bx"]), IV.exact(b["by"])
    th_iv = th_deg_iv * DEG2RAD
    c_iv = IV.cos(th_iv)
    s_iv = IV.sin(th_iv)
    t_steps = IV(k - 1) + al_iv
    grav_steps = IV(0.5 * (k - 1) * k) + al_iv * IV(k)
    xk_iv = IV(7.2) + t_steps * V0_DT_IV * c_iv
    yk_iv = IV(1.8) + t_steps * V0_DT_IV * s_iv + grav_steps * G_DT2_IV
    pA = [(xk_iv + c_iv * vx - s_iv * vy, yk_iv + s_iv * vx + c_iv * vy) for vx, vy in vA_iv]
    pB = [(bx_iv + vx, by_iv + vy) for vx, vy in vB_iv]
    if kind == "B":
        nx, ny = nB_iv[idx]
        vBx, vBy = pB[idx]
        dots = [(px - vBx) * nx + (py - vBy) * ny for px, py in pA]
        return IV.min_list(dots)
    else:
        nlx, nly = nA_iv[idx]
        nx = c_iv * nlx - s_iv * nly
        ny = s_iv * nlx + c_iv * nly
        vAx, vAy = pA[idx]
        dots = [(px - vAx) * nx + (py - vAy) * ny for px, py in pB]
        return IV.min_list(dots)

def iv_eval_wall19(k: int, al_iv: IV, th_deg_iv: IV) -> IV:
    th_iv = th_deg_iv * DEG2RAD
    c_iv = IV.cos(th_iv)
    s_iv = IV.sin(th_iv)
    t_steps = IV(k - 1) + al_iv
    xk_iv = IV(7.2) + t_steps * V0_DT_IV * c_iv
    dots = [xk_iv + c_iv * vx - s_iv * vy for vx, vy in vA_iv]
    return IV.min_list(dots)

def iv_alpha_target(uid: int, k: int, th_deg_iv: IV, kind: str, idx: int, target: float = 0.0075) -> IV:
    s0 = iv_eval_axis(uid, k, IV(0.0), th_deg_iv, kind, idx)
    s1 = iv_eval_axis(uid, k, IV(1.0), th_deg_iv, kind, idx)
    return (s0 - IV(target)) / (s0 - s1)

def iv_alpha_wall19(k: int, th_deg_iv: IV, target: float = 0.0075) -> IV:
    s0 = iv_eval_wall19(k, IV(0.0), th_deg_iv)
    s1 = iv_eval_wall19(k, IV(1.0), th_deg_iv)
    return (s0 - IV(target)) / (s0 - s1)

def iv_d_alpha_d_theta_rad(uid: int, k: int, th_deg_iv: IV, kind: str, idx: int, v_idx: int, target: float = 0.0075) -> IV:
    """Outward-rounded interval enclosure of d(alpha)/d(theta_rad) = (s0'(theta) - alpha(theta)*Delta_s'(theta)) / Delta_s(theta),
    plus verification that vertex v_idx is the unique minimizing vertex across th_deg_iv.
    """
    b = next(x for x in BLOCKS_SEED3000 if x["uid"] == uid)
    bx_iv, by_iv = IV.exact(b["bx"]), IV.exact(b["by"])
    th_iv = th_deg_iv * DEG2RAD
    c_iv = IV.cos(th_iv)
    s_iv = IV.sin(th_iv)
    al_iv = iv_alpha_target(uid, k, th_deg_iv, kind, idx, target)
    s0_iv = iv_eval_axis(uid, k, IV(0.0), th_deg_iv, kind, idx)
    s1_iv = iv_eval_axis(uid, k, IV(1.0), th_deg_iv, kind, idx)
    ds_iv = s0_iv - s1_iv

    # Verify v_idx is strictly minimal across th_deg_iv
    if kind == "B":
        nx, ny = nB_iv[idx]
        projs = [(c_iv * vx - s_iv * vy) * nx + (s_iv * vx + c_iv * vy) * ny for vx, vy in vA_iv]
        for i in range(8):
            if i != v_idx:
                assert projs[v_idx].hi < projs[i].lo, f"Vertex {v_idx} not unique min vs {i}"
        vx, vy = vA_iv[v_idx]
        # pA_v(al, th) = (x0 + (k-1+al)*V0*DT*cos(th) + cos(th)*vx - sin(th)*vy, y0 + (k-1+al)*V0*DT*sin(th) + grav + sin(th)*vx + cos(th)*vy)
        # d(pA_v)/d(th) at al=0:
        dx0_dth = -IV(k - 1) * V0_DT_IV * s_iv - s_iv * vx - c_iv * vy
        dy0_dth = IV(k - 1) * V0_DT_IV * c_iv + c_iv * vx - s_iv * vy
        s0_prime = dx0_dth * nx + dy0_dth * ny
        # Delta_s(th) = s0(th) - s1(th) = -V0*DT*(cos(th)*nx + sin(th)*ny) - G*DT^2*k*ny
        ds_prime = -V0_DT_IV * (-s_iv * nx + c_iv * ny)
        return (s0_prime - al_iv * ds_prime) / ds_iv
    else:
        nlx, nly = nA_iv[idx]
        nx = c_iv * nlx - s_iv * nly
        ny = s_iv * nlx + c_iv * nly
        projs = [vx * nx + vy * ny for vx, vy in vB_iv]
        for j in range(8):
            if j != v_idx:
                assert projs[v_idx].hi < projs[j].lo, f"Vertex {v_idx} not unique min vs {j}"
        # For ball-face axis idx: n_A(th) = R(th)*n_local, and (pB_j - pA_idx) . n_A(th) = (pB_j - x_k(al, th)) . n_A(th) - d_{A,idx}
        # where d_{A,idx} = vA_{idx} . n_local is independent of theta!
        # Let D_0(th) = pB_v - x_k(0, th):
        x0_iv = IV(7.2) + IV(k - 1) * V0_DT_IV * c_iv
        y0_iv = IV(1.8) + IV(k - 1) * V0_DT_IV * s_iv + IV(0.5 * (k - 1) * k) * G_DT2_IV
        Dx0 = (bx_iv + vB_iv[v_idx][0]) - x0_iv
        Dy0 = (by_iv + vB_iv[v_idx][1]) - y0_iv
        dDx0_dth = IV(k - 1) * V0_DT_IV * s_iv
        dDy0_dth = -IV(k - 1) * V0_DT_IV * c_iv
        dnx_dth = -ny
        dny_dth = nx
        s0_prime = dDx0_dth * nx + dDy0_dth * ny + Dx0 * dnx_dth + Dy0 * dny_dth
        # Notice: (x_k(1, th) - x_k(0, th)) . n_A(th) = V0*DT*(cos(th)*nx + sin(th)*ny) + k*G*DT^2*ny
        # And cos(th)*nx + sin(th)*ny = nlx is CONSTANT in theta!
        # So Delta_s(th) = s0(th) - s1(th) = V0*DT*nlx + k*G*DT^2*ny(th), whose theta-derivative is simply k*G*DT^2*dny_dth!
        ds_prime = IV(k) * G_DT2_IV * dny_dth
        return (s0_prime - al_iv * ds_prime) / ds_iv

def iv_d_alpha_wall19_d_theta_rad(k: int, th_deg_iv: IV, v_idx: int = 4, target: float = 0.0075) -> IV:
    th_iv = th_deg_iv * DEG2RAD
    c_iv = IV.cos(th_iv)
    s_iv = IV.sin(th_iv)
    projs = [c_iv * vx - s_iv * vy for vx, vy in vA_iv]
    for i in range(8):
        if i != v_idx:
            assert projs[v_idx].hi < projs[i].lo, f"Wall vertex {v_idx} not unique min vs {i}"
    vx, vy = vA_iv[v_idx]
    al_iv = iv_alpha_wall19(k, th_deg_iv, target)
    s0_iv = iv_eval_wall19(k, IV(0.0), th_deg_iv)
    s1_iv = iv_eval_wall19(k, IV(1.0), th_deg_iv)
    ds_iv = s0_iv - s1_iv
    s0_prime = -IV(k - 1) * V0_DT_IV * s_iv - s_iv * vx - c_iv * vy
    # Delta_s(th) = s0(th) - s1(th) = -V0*DT*cos(th), so ds_prime = +V0*DT*sin(th)
    ds_prime = V0_DT_IV * s_iv
    return (s0_prime - al_iv * ds_prime) / ds_iv

def iv_min_alpha_max2(uid: int, k: int, th_deg_iv: IV, ax1, ax2) -> float:
    s1_0 = iv_eval_axis(uid, k, IV(0.0), th_deg_iv, *ax1)
    s1_1 = iv_eval_axis(uid, k, IV(1.0), th_deg_iv, *ax1)
    s2_0 = iv_eval_axis(uid, k, IV(0.0), th_deg_iv, *ax2)
    s2_1 = iv_eval_axis(uid, k, IV(1.0), th_deg_iv, *ax2)
    a0, a1 = s1_0.lo, s1_1.lo
    b0, b1 = s2_0.lo, s2_1.lo
    d0 = a0 - b0
    d1 = a1 - b1
    cands = [max(a0, b0), max(a1, b1)]
    if d0 * d1 < 0.0:
        al_iv = IV(down(d0), up(d0)) / (IV(down(d0), up(d0)) - IV(down(d1), up(d1)))
        val_iv = IV(a0) + al_iv * (IV(a1) - IV(a0))
        cands.append(val_iv.lo)
    return min(cands)
